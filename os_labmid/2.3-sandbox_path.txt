/* ============================================================================
 * STEP 1 ---------------------------------------------------------------------
 *   FILE   kernel/proc.h
 *   WHERE  inside struct proc, right under  int interpose_mask;
 *   WHY    each process must remember its own allowed path.
 *          MAXPATH (128) comes from kernel/param.h (already included).
 * ============================================================================ */

  char interpose_path[MAXPATH];   /* the one path still allowed ("-" = none) */


/* ============================================================================
 * STEP 2 ---------------------------------------------------------------------
 *   FILE   kernel/sysproc.c
 *   WHERE  REPLACE the whole sys_interpose() from Task 2 with this.
 *   WHY    argument 1 is a POINTER into user memory. The kernel cannot just
 *          read it (different page table), so argstr() copies the string
 *          safely into a kernel buffer. Returns < 0 if the pointer is bad.
 *          safestrcpy() always NUL-terminates.
 * ============================================================================ */

uint64
sys_interpose(void)
{
  int mask;
  char path[MAXPATH];
  struct proc *p = myproc();

  argint(0, &mask);                       /* arg 0 -> mask */
  if(argstr(1, path, MAXPATH) < 0)        /* arg 1 -> path string */
    return -1;

  p->interpose_mask = mask;
  safestrcpy(p->interpose_path, path, MAXPATH);
  return 0;
}


/* ============================================================================
 * STEP 3 ---------------------------------------------------------------------
 *   FILE   kernel/proc.c
 *   WHERE  in the fork function (kfork() or fork()),
 *          right AFTER your Task 2 line:
 *              np->interpose_mask = p->interpose_mask;
 *   WHY    INHERITANCE of the path. np = child, p = parent.
 * ============================================================================ */

  safestrcpy(np->interpose_path, p->interpose_path, MAXPATH);


/* ============================================================================
 * STEP 4 ---------------------------------------------------------------------
 *   FILE   kernel/syscall.c
 *   WHERE  ABOVE the function syscall(), and BELOW the function argstr()
 *          (it calls argstr, so argstr must come first in the file).
 *   WHY    decides whether a banned call is let through anyway.
 *          open(path, ...) and exec(path, ...) BOTH take the path as
 *          argument 0, so argstr(0, ...) reads it for either one.
 * ============================================================================ */

/* Return 1 if this is an open/exec whose path equals the allowed path. */
static int
path_allowed(int num, struct proc *p)
{
  char path[MAXPATH];

  if(num != SYS_open && num != SYS_exec)      /* only open/exec have a path */
    return 0;
  if(p->interpose_path[0] == '-' && p->interpose_path[1] == 0)
    return 0;                                 /* "-" means nothing allowed */
  if(argstr(0, path, MAXPATH) < 0)            /* read the call's path arg */
    return 0;
  return strncmp(path, p->interpose_path, MAXPATH) == 0;
}


/* ============================================================================
 * STEP 5 ---------------------------------------------------------------------
 *   FILE   kernel/syscall.c, inside syscall()
 *   WHERE  REPLACE the Task 2 check
 *              if((p->interpose_mask >> num) & 1)
 *   WHY    reject only if banned AND not the allowed path.
 *   NOTE   path_allowed() runs BEFORE a0 is overwritten with the return
 *          value, so argument 0 (a0) still holds the path pointer.
 * ============================================================================ */

    if(((p->interpose_mask >> num) & 1) && !path_allowed(num, p))
      p->trapframe->a0 = -1;                 /* banned: fail */
    else
      p->trapframe->a0 = syscalls[num]();    /* allowed: run it */


/* ============================================================================
 *  FINAL syscall() FOR REFERENCE  (what the whole function looks like now)
 * ============================================================================ */

void
syscall(void)
{
  int num;
  struct proc *p = myproc();

  num = p->trapframe->a7;
  if(num > 0 && num < NELEM(syscalls) && syscalls[num]) {
    if(((p->interpose_mask >> num) & 1) && !path_allowed(num, p))
      p->trapframe->a0 = -1;
    else
      p->trapframe->a0 = syscalls[num]();
  } else {
    printf("%d %s: unknown sys call %d\n", p->pid, p->name, num);
    p->trapframe->a0 = -1;
  }
}
